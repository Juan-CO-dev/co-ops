```ts
// lib/guides/readability-shared.ts
/**
 * readability — the Guides v2 plain-language line (Juan 2026-10-13: "6th grade reading level").
 *
 * PURE. Flesch-Kincaid grade = 0.39·(words/sentence) + 11.8·(syllables/word) − 15.59.
 * The syllable count is the usual English heuristic (vowel groups, minus a silent final e), which
 * is close enough to hold a line; it is not a linguistics library. Bold UI labels count as one
 * short word ("X"), because a label is read off the screen, not decoded. Headings, photos, tables
 * and code are left out. Only the English is scored; the Spanish is read by a reviewer.
 */

export const SPANISH_HEADING = /^###\s+En español\s*$/im;

export function syllables(word: string): number {
  if (/^[0-9]/.test(word)) return 1;
  const w = word.toLowerCase().replace(/[^a-z]/g, "");
  if (!w) return 0;
  if (w.length <= 3) return 1;
  const trimmed = w.replace(/(?:[^laeiouy]es|ed|[^laeiouy]e)$/, "").replace(/^y/, "");
  const groups = trimmed.match(/[aeiouy]{1,2}/g);
  return Math.max(1, groups ? groups.length : 0);
}

function inlinePlain(line: string): string {
  return line
    .replace(/\*\*[^*]+\*\*/g, "X")
    .replace(/`[^`]+`/g, "X")
    .replace(/!\[[^\]]*\]\([^)]*\)/g, "")
    .replace(/\[([^\]]+)\]\([^)]*\)/g, "$1")
    .replace(/[*_]([^*_]+)[*_]/g, "$1")
    .trim();
}

const endWithStop = (s: string) => (/[.!?:]$/.test(s) ? s.replace(/:$/, ".") : `${s}.`);

export function plainSentences(markdown: string): string[] {
  const out: string[] = [];
  let paragraph: string[] = [];
  let inCode = false;
  const flush = () => {
    const text = paragraph.join(" ").trim();
    paragraph = [];
    if (!text) return;
    for (const s of text.split(/(?<=[.!?])\s+/)) if (/[A-Za-z0-9]/.test(s)) out.push(endWithStop(s.trim()));
  };
  for (const raw of markdown.split(/\r?\n/)) {
    const line = raw.trim();
    if (line.startsWith("```")) { flush(); inCode = !inCode; continue; }
    if (inCode) continue;
    if (!line || /^#/.test(line) || /^!\[/.test(line) || /^\|/.test(line) || /^-{3,}$/.test(line) || /^<!--/.test(line)) { flush(); continue; }
    const bullet = line.match(/^(?:[-*+]|\d+\.)\s+(.*)$/);
    if (bullet) { flush(); paragraph.push(inlinePlain(bullet[1] ?? "")); flush(); continue; }
    paragraph.push(inlinePlain(line));
  }
  flush();
  return out;
}

export function fkGrade(sentences: string[]): { grade: number; words: number; sentences: number; syllables: number } {
  const words = sentences.flatMap((s) => s.match(/[A-Za-z0-9][A-Za-z0-9'’:-]*/g) ?? []);
  if (words.length === 0 || sentences.length === 0) return { grade: 0, words: 0, sentences: sentences.length, syllables: 0 };
  const syl = words.reduce((n, w) => n + syllables(w), 0);
  const grade = 0.39 * (words.length / sentences.length) + 11.8 * (syl / words.length) - 15.59;
  return { grade, words: words.length, sentences: sentences.length, syllables: syl };
}

export function englishSections(markdown: string): { heading: string; body: string }[] {
  const parts = markdown.split(/^## /m).slice(1);
  return parts.map((part) => {
    const nl = part.indexOf("\n");
    const heading = (nl === -1 ? part : part.slice(0, nl)).trim();
    const rest = nl === -1 ? "" : part.slice(nl);
    const es = rest.search(SPANISH_HEADING);
    return { heading, body: es === -1 ? rest : rest.slice(0, es) };
  });
}
```

```js
// scripts/guide-test/browser-daemon.mjs
/** Guides v2 tester browser. SIM ONLY: every page lives on http://localhost:3100 and nothing else.
 * Run: node scripts/guide-test/browser-daemon.mjs        (listens on 127.0.0.1:3199)
 * Every tester family (Claude subagents, Codex, the GLM/DeepSeek API loop) drives the app through
 * this one daemon (via gt.mjs or plain HTTP), so the comparison is fair and each run has its own
 * browser context. Testers may only look, tap, type, press, scroll and go back; `goto` and `shot`
 * exist for photo walkers only (sessions started with walker=true).
 */
import http from "node:http";
import fs from "node:fs";
import path from "node:path";
import { chromium } from "playwright";

const SIM = "http://localhost:3100";
const PORT = 3199;
const RUNS = path.resolve("docs/guides/testing/runs");
const IMG = path.resolve("docs/guides/img");
const SHOT_FILE = /^(staff|manager|catering)\/[0-9]{2}[a-z]?-[a-z0-9-]+\.png$/;
const VIEWPORTS = {
  phone: { viewport: { width: 390, height: 844 }, deviceScaleFactor: 2, isMobile: true, hasTouch: true },
  tablet: { viewport: { width: 1024, height: 768 }, deviceScaleFactor: 2, hasTouch: true },
  desktop: { viewport: { width: 1440, height: 900 }, deviceScaleFactor: 1 },
};
const CONTROLS = 'a[href], button, input:not([type=hidden]), select, textarea, [role=button], [role=link], [role=tab], [role=checkbox], [role=radio], [role=switch], [role=menuitem], [role=option], summary';

const browser = await chromium.launch({ headless: true });
const sessions = new Map();

function log(s, entry) {
  fs.appendFileSync(path.join(s.dir, "actions.jsonl"), JSON.stringify({ t: new Date().toISOString(), ...entry }) + "\n");
}

async function settle(page) {
  await page.waitForLoadState("networkidle", { timeout: 5000 }).catch(() => {});
  await page.waitForTimeout(300);
}

async function start({ session, viewport = "phone", walker = false }) {
  if (sessions.has(session)) await stop({ session });
  const dir = path.join(RUNS, session);
  fs.mkdirSync(dir, { recursive: true });
  const ctx = await browser.newContext({ ...(VIEWPORTS[viewport] ?? VIEWPORTS.phone), baseURL: SIM });
  await ctx.route("**/*", (route) => {
    const req = route.request();
    if (req.isNavigationRequest() && !req.url().startsWith(SIM)) return route.abort();
    return route.continue();
  });
  const page = await ctx.newPage();
  const s = { ctx, page, dir, walker: Boolean(walker), looks: 0, dialogs: [] };
  page.on("dialog", async (d) => { s.dialogs.push(`${d.type()}: ${d.message()}`); await d.accept().catch(() => {}); });
  sessions.set(session, s);
  await page.goto("/");
  await settle(page);
  log(s, { action: "start", viewport, walker: s.walker });
  return `OK started ${session} (${viewport}). Now run: look`;
}

async function look({ session }) {
  const s = must(session);
  const { page } = s;
  const controls = await page.$$eval(CONTROLS, (nodes) => {
    let i = 0;
    const out = [];
    for (const n of nodes) {
      n.removeAttribute("data-gt");
      const r = n.getBoundingClientRect();
      const st = getComputedStyle(n);
      if (r.width === 0 || r.height === 0 || st.visibility === "hidden" || st.display === "none" || n.closest('[aria-hidden="true"]')) continue;
      i += 1;
      n.setAttribute("data-gt", String(i));
      const tag = n.tagName.toLowerCase();
      const kind = tag === "input" ? `input:${n.type}` : (n.getAttribute("role") ?? tag);
      const label = (n.getAttribute("aria-label") || n.innerText || n.getAttribute("placeholder") || n.getAttribute("title") || n.getAttribute("name") || "").replace(/\s+/g, " ").trim().slice(0, 90);
      let extra = "";
      if (tag === "input" || tag === "textarea") extra = n.type === "checkbox" || n.type === "radio" ? (n.checked ? " [checked]" : " [unchecked]") : ` value="${String(n.value).slice(0, 40)}"`;
      if (tag === "select") extra = ` selected="${n.selectedOptions[0]?.text ?? ""}" options=[${[...n.options].map((o) => o.text).slice(0, 15).join(" | ")}]`;
      if (n.disabled || n.getAttribute("aria-disabled") === "true") extra += " [disabled]";
      out.push(`[${i}] ${kind} "${label}"${extra}`);
    }
    return out;
  });
  const text = (await page.evaluate(() => document.body.innerText)).replace(/\n{3,}/g, "\n\n").trim().slice(0, 6000);
  s.looks += 1;
  const shot = path.join(s.dir, `${String(s.looks).padStart(3, "0")}.png`);
  await page.screenshot({ path: shot });
  const dialogs = s.dialogs.splice(0).map((d) => `DIALOG (accepted) ${d}`).join("\n");
  log(s, { action: "look", url: page.url() });
  return [`URL: ${page.url().replace(SIM, "")}`, `TITLE: ${await page.title()}`, dialogs, "SCREEN TEXT:", text, "", "CONTROLS:", ...controls, "", `SCREENSHOT: ${shot}`].filter((l) => l !== "").join("\n");
}

function locate(page, target) {
  const t = String(target ?? "").trim();
  if (/^#?\d+$/.test(t)) return page.locator(`[data-gt="${t.replace("#", "")}"]`);
  return page.getByRole("button", { name: t }).or(page.getByRole("link", { name: t })).or(page.getByRole("tab", { name: t })).or(page.getByText(t, { exact: true })).first();
}

async function act(session, entry, fn) {
  const s = must(session);
  try {
    await fn(s.page);
  } catch (e) {
    log(s, { ...entry, error: String(e.message).slice(0, 200) });
    return `FAILED: ${String(e.message).split("\n")[0].slice(0, 200)}. The screen may have changed: run look again.`;
  }
  await settle(s.page);
  log(s, { ...entry, url: s.page.url() });
  return `OK. Now at ${s.page.url().replace(SIM, "")}. Run look to see the screen.`;
}

const tap = ({ session, target }) => act(session, { action: "tap", target }, (p) => locate(p, target).click({ timeout: 5000 }));
const type = ({ session, target, text }) => act(session, { action: "type", target, text }, async (p) => {
  const el = locate(p, target);
  const tag = await el.evaluate((n) => n.tagName.toLowerCase(), undefined, { timeout: 5000 });
  if (tag === "select") await el.selectOption({ label: String(text) }, { timeout: 5000 });
  else await el.fill(String(text), { timeout: 5000 });
});
const press = ({ session, key }) => act(session, { action: "press", key }, (p) => p.keyboard.press(String(key)));
const scroll = ({ session, direction = "down" }) => act(session, { action: "scroll", direction }, (p) => p.mouse.wheel(0, direction === "up" ? -700 : 700));
const back = ({ session }) => act(session, { action: "back" }, (p) => p.goBack());

async function goto({ session, path: to }) {
  const s = must(session);
  if (!s.walker) return "REFUSED: testers move by tapping, like a person. goto is for photo walkers only.";
  if (!String(to).startsWith("/")) return "REFUSED: goto takes a path that starts with /.";
  return act(session, { action: "goto", path: to }, (p) => p.goto(String(to)));
}

async function shot({ session, file, fullPage = false }) {
  const s = must(session);
  if (!s.walker) return "REFUSED: shot is for photo walkers only.";
  if (!SHOT_FILE.test(String(file))) return "REFUSED: file must look like staff/07-opening-verify.png";
  const out = path.join(IMG, String(file));
  fs.mkdirSync(path.dirname(out), { recursive: true });
  await s.page.screenshot({ path: out, fullPage: Boolean(fullPage) });
  log(s, { action: "shot", file });
  return `OK saved docs/guides/img/${file}`;
}

async function stop({ session }) {
  const s = sessions.get(session);
  if (!s) return "OK (no such session)";
  sessions.delete(session);
  await s.ctx.close().catch(() => {});
  return `OK stopped ${session}`;
}

function must(session) {
  const s = sessions.get(session);
  if (!s) throw new Error(`no session "${session}": run start first`);
  return s;
}

const ROUTES = { start, look, tap, type, press, scroll, back, goto, shot, stop };

http.createServer((req, res) => {
  const name = (req.url ?? "/").slice(1);
  let body = "";
  req.on("data", (c) => { body += c; });
  req.on("end", async () => {
    const fn = ROUTES[name];
    if (req.method !== "POST" || !fn) { res.writeHead(404); return res.end("unknown command"); }
    try {
      const out = await fn(JSON.parse(body || "{}"));
      res.writeHead(200, { "content-type": "text/plain; charset=utf-8" });
      res.end(out);
    } catch (e) {
      res.writeHead(400, { "content-type": "text/plain; charset=utf-8" });
      res.end(`ERROR: ${String(e.message).slice(0, 300)}`);
    }
  });
}).listen(PORT, "127.0.0.1", () => console.log(`guide-test daemon on http://127.0.0.1:${PORT} (sim ${SIM})`));
```

/** The one tool every guide tester uses. No dependencies: plain Node fetch to the daemon.
 * Usage: node gt.mjs <session> <command> [args]
 *   start [phone|tablet|desktop] [walker] · look · tap <number|label> · type <number|label> <text>
 *   press <key> · scroll [down|up] · back · stop · goto </path> (walkers) · shot <guide/NN-slug.png> [full] (walkers)
 */
const [session, cmd, ...args] = process.argv.slice(2);
if (!session || !cmd) { console.log("usage: node gt.mjs <session> <command> [args]"); process.exit(2); }
const body = { session };
if (cmd === "start") { body.viewport = args[0] ?? "phone"; body.walker = args[1] === "walker"; }
if (cmd === "tap") body.target = args.join(" ");

```python
# scripts/guide-test/api_tester.py
"""Guides v2 tester loop for API-only models (GLM and DeepSeek). The model gets the same tools as every other
tester (look/tap/type/press/scroll/back), executed against the guide-test daemon, and ends with `finish(report)`.
Keys are read by NAME from the Hermes .env and never printed.
Usage:
  python scripts/guide-test/api_tester.py --list-models glm|deepseek
  python scripts/guide-test/api_tester.py --provider glm --model glm-5.3 --workspace <dir> --session <id> --out <report.md>
"""
import argparse, json, os, sys, time, urllib.request

ENV = os.path.expanduser("~/AppData/Local/hermes/.env")
PROVIDERS = {
    "glm": ("GLM_API_KEY", "https://api.z.ai/api/paas/v4"),
    "deepseek": ("DEEPSEEK_API_KEY", "https://api.deepseek.com/v1"),
}
DAEMON = "http://127.0.0.1:3199"
MAX_STEPS = 300

def key(name):
    with open(ENV, encoding="utf-8") as f:
        for line in f:
            if line.startswith(name + "="):
                return line.strip().split("=", 1)[1].strip('"').strip("'")
    raise RuntimeError(f"{name} not in the Hermes .env")

def http_json(url, payload=None, auth=None, timeout=600):
    data = None if payload is None else json.dumps(payload).encode()
    headers = {"Content-Type": "application/json"}
    if auth: headers["Authorization"] = f"Bearer {auth}"
    req = urllib.request.Request(url, data=data, headers=headers, method="POST" if data else "GET")
    with urllib.request.urlopen(req, timeout=timeout) as r:
        return json.loads(r.read().decode("utf-8"))

def daemon(cmd, **body):
    req = urllib.request.Request(f"{DAEMON}/{cmd}", data=json.dumps(body).encode(), headers={"Content-Type": "application/json"}, method="POST")
    try:
        with urllib.request.urlopen(req, timeout=60) as r:
            return r.read().decode("utf-8")
    except urllib.error.HTTPError as e:
        return e.read().decode("utf-8")

def tool(name, desc, props, required):
    return {"type": "function", "function": {"name": name, "description": desc, "parameters": {"type": "object", "properties": props, "required": required}}}

TOOLS = [
    tool("look", "See the screen: its text, numbered controls, and a screenshot path.", {}, []),
    tool("tap", "Tap a control by its number from the last look, or by its exact label.", {"target": {"type": "string"}}, ["target"]),
    tool("type", "Fill a field (or pick an option in a dropdown) by number or label.", {"target": {"type": "string"}, "text": {"type": "string"}}, ["target", "text"]),
    tool("press", "Press a key: Enter, Tab or Escape.", {"key": {"type": "string"}}, ["key"]),
    tool("scroll", "Scroll the screen.", {"direction": {"type": "string", "enum": ["down", "up"]}}, ["direction"]),
    tool("back", "Go back one screen.", {}, []),
    tool("finish", "End the test and hand in your report (Markdown, in the shape the instructions give).", {"report": {"type": "string"}}, ["report"]),
]

def shrink_old_looks(messages, keep_last=3):
    looks = [i for i, m in enumerate(messages) if m.get("role") == "tool" and m.get("name") == "look"]
    for i in looks[:-keep_last]:
        messages[i]["content"] = messages[i]["content"][:300] + "\n…(older screen, trimmed)"

def run(args):
    kname, base = PROVIDERS[args.provider]
    prompt = open(os.path.join(args.workspace, "PROMPT.md"), encoding="utf-8").read()
    guide = open(os.path.join(args.workspace, "guide.md"), encoding="utf-8").read()
    messages = [
        {"role": "system", "content": prompt},
        {"role": "user", "content": "Here is the guide. Read it, then start.\n\n" + guide},
    ]
    daemon("start", session=args.session, viewport=args.viewport)
    report, steps, t0 = None, 0, time.time()
    while steps < MAX_STEPS and report is None:
        shrink_old_looks(messages)
        payload = {"model": args.model, "messages": messages, "tools": TOOLS, "temperature": 0.2, "max_tokens": 4000}
        try:
            data = http_json(f"{base}/chat/completions", payload, auth=key(kname))
        except Exception as e:
            print(f"model call failed: {type(e).__name__}: {str(e)[:200]}", file=sys.stderr); time.sleep(10); continue
        msg = data["choices"][0]["message"]
        messages.append({k: v for k, v in msg.items() if k in ("role", "content", "tool_calls", "reasoning_content")})
        calls = msg.get("tool_calls") or []
        if not calls:
            messages.append({"role": "user", "content": "Use a tool. When you are done with every goal, call finish with your report."})
            continue
        for c in calls:
            steps += 1
            name = c["function"]["name"]
            try: a = json.loads(c["function"].get("arguments") or "{}")
            except json.JSONDecodeError: a = {}
            if name == "finish":
                report = a.get("report", "")
                out = "Report received."
            elif name in ("look", "tap", "type", "press", "scroll", "back"):
                out = daemon(name, session=args.session, **a)
            else:
                out = f"Unknown tool {name}."
            messages.append({"role": "tool", "tool_call_id": c["id"], "name": name, "content": out})
    daemon("stop", session=args.session)
    header = f"<!-- tester {args.provider}/{args.model} · session {args.session} · {steps} actions · {round(time.time() - t0)}s -->\n"
    body = report if report is not None else f"# Report — {args.model}\n\nRUN ENDED WITHOUT A REPORT after {steps} actions (limit {MAX_STEPS})."
    os.makedirs(os.path.dirname(args.out), exist_ok=True)
    with open(args.out, "w", encoding="utf-8", newline="\n") as f:
        f.write(header + body + "\n")
    print(f"{args.model} {args.session}: {steps} actions -> {args.out}")

def list_models(provider):
    kname, base = PROVIDERS[provider]
    data = http_json(f"{base}/models", auth=key(kname))
    for m in data.get("data", []):
        print(m.get("id"))

if __name__ == "__main__":
    p = argparse.ArgumentParser()
    p.add_argument("--list-models", choices=list(PROVIDERS))
    p.add_argument("--provider", choices=list(PROVIDERS))
    p.add_argument("--model"); p.add_argument("--workspace"); p.add_argument("--session"); p.add_argument("--out")
    p.add_argument("--viewport", default="phone", choices=["phone", "tablet", "desktop"])
    a = p.parse_args()
    if a.list_models: list_models(a.list_models)
    else: run(a)
```

// tests/guides-reading-level.test.ts
import { describe, it, expect } from "vitest";
import { readFileSync } from "node:fs";
import { englishSections, fkGrade, plainSentences } from "@/lib/guides/readability-shared";

const MAX_GRADE = 6;
const MIN_WORDS = 30; // the formula swings wildly on a few words

describe("guides read at grade 6 or lower (Juan 2026-10-13)", () => {
  it.each(["staff", "manager", "catering"])("%s guide: every English section", (g) => {
    const md = readFileSync(`docs/guides/${g}-guide.md`, "utf8");
    const tooHard = englishSections(md).flatMap(({ heading, body }) => {
      const sentences = plainSentences(body);
      const r = fkGrade(sentences);
      if (r.words < MIN_WORDS || r.grade <= MAX_GRADE) return [];
      const longest = [...sentences].sort((a, b) => b.split(" ").length - a.split(" ").length).slice(0, 3);
      return [`${heading}: grade ${r.grade.toFixed(1)} — longest: ${longest.join(" | ")}`];
    });
    expect(tooHard).toEqual([]);
  });
});
```
